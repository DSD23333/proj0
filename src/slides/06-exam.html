					void __am_gpu_fbdraw(AM_GPU_FBDRAW_T *ctl) {

						if(ctl->pixels != NULL){
							uint32_t *pixels = ctl->pixels;
							int count = 0;

							for(int yi = ctl->y; yi < ctl->y + ctl->h; yi++){
								for(int xi = ctl->x; xi <ctl->x + ctl->w; xi++){
									outl(FB_ADDR + (yi * win_size_w + xi) * 4, pixels[count]); //NOTE: debug: * 4
									count ++;
								}
							}

						}

						if (ctl->sync) {
							outl(SYNC_ADDR, 1);
						}
					}
